   8|     def get(self, request):
   9|         """Returns currently published file."""
  10|         document_model = self.model.objects.published()
  11|         if document_model is None:
  12|             raise Http404
  13|         file = document_model.file_obj
  14|         if file is None:
  15|             raise Http404
  16|         filename = document_model.filename()
  17|         if filename.endswith('.html'):
  18|             response = FileResponse(file, as_attachment=False,
  19|                                     content_type='text/html')
  20|         elif filename.endswith('.pdf'):
  21|             response = FileResponse(file, as_attachment=False,
  22|                                     content_type='application/pdf')
